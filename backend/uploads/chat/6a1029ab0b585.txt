   INPUT

$token    = trim($data['token'] ?? '');

    sendError("Token and password required", 400, "VALIDATION_ERROR");

/* -------------------------------------------------
   PASSWORD VALIDATION
------------------------------------------------- */
if (strlen($password) < 8) {
    sendError(
        "Password must be at least 8 characters",

        "WEAK_PASSWORD"
    );
}

/* -------------------------------------------------
   VALIDATE INVITE TOKEN
------------------------------------------------- */
$stmt = $pdo->prepare("
    SELECT ui.user_id, u.status
    FROM user_invites ui
    JOIN users u ON u.id = ui.user_id
    WHERE ui.token = ?
      AND ui.used_at IS NULL
      AND ui.expires_at > NOW()
    LIMIT 1
");
$stmt->execute([$token]);
$invite = $stmt->fetch(PDO::FETCH_ASSOC);

if (!$invite) {
    sendError(
        "Invalid or expired activation link",
        400,
        "INVALID_TOKEN"
    );
}

/* -------------------------------------------------
   CHECK USER STATUS
------------------------------------------------- */
if ($invite['status'] === 'ACTIVE') {
    sendError(
        "Account already activated",
        400,
        "ALREADY_ACTIVATED"
    );
}

/* -------------------------------------------------
   TRANSACTION
------------------------------------------------- */
try {
    $pdo->beginTransaction();

    /* HASH PASSWORD */
    $hashedPassword = password_hash($password, PASSWORD_DEFAULT);

    /* ACTIVATE USER */
    $stmt = $pdo->prepare("
        UPDATE users
        SET password = ?,
            status = 'ACTIVE',
            email_verified = 1,
            updated_at = NOW()
        WHERE id = ?
    ");
    $stmt->execute([$hashedPassword, $invite['user_id']]);

    /* MARK INVITE USED */
    $stmt = $pdo->prepare("
        UPDATE user_invites
        SET used_at = NOW()
        WHERE token = ?
    ");
    $stmt->execute([$token]);

    $pdo->commit();

    sendSuccess("Account activated successfully");

} catch (Throwable $e) {
    $pdo->rollBack();

    sendError(
        "Failed to activate account",
        500,
        "SERVER_ERROR"
    );
}

middleware.php
<?php
require_once __DIR__ . '/../config/config.php';
require_once __DIR__ . '/../helpers/db.php';
require_once __DIR__ . '/../helpers/jwt.php';
require_once __DIR__ . '/../helpers/response.php';

/*
|--------------------------------------------------------------------------
| REQUIRE AUTH (JWT)
|--------------------------------------------------------------------------
| - Validates Bearer token
| - Decodes JWT
| - Sets $GLOBALS['auth_user'] = ['id','role']
|--------------------------------------------------------------------------
*/
function require_auth(): void
{
    $headers = getallheaders();

    if (empty($headers['Authorization'])) {
        sendError("Authorization header missing", 401, "AUTH_HEADER_MISSING");
    }

    if (!preg_match('/Bearer\s+(\S+)/', $headers['Authorization'], $matches)) {
        sendError("Invalid authorization header format", 401, "INVALID_AUTH_HEADER");
    }

    $token = $matches[1];
    $payload = jwt_decode_token($token);

    if (!$payload) {
        sendError("Invalid or expired token", 401, "INVALID_TOKEN");
    }

    if (empty($payload->id) || empty($payload->role)) {
        sendError("Invalid token payload", 401, "INVALID_TOKEN_PAYLOAD");
    }

    // Set authenticated user globally
    $GLOBALS['auth_user'] = [
        'id'   => (int) $payload->id,
        'role' => $payload->role
    ];
}

/*
|--------------------------------------------------------------------------
| REQUIRE ADMIN
|--------------------------------------------------------------------------
| Usage:
|   require_admin();
|--------------------------------------------------------------------------
*/
function require_admin(): void
{
    require_auth();

    if ($GLOBALS['auth_user']['role'] !== 'admin') {
        sendError("Admin access only", 403, "FORBIDDEN");
    }
}

/*
|--------------------------------------------------------------------------
| REQUIRE BUSINESS
|--------------------------------------------------------------------------
| Usage:
|   require_business();
|--------------------------------------------------------------------------
*/
function require_business(): void
{
    require_auth();

    if ($GLOBALS['auth_user']['role'] !== 'business') {
        sendError("Business access only", 403, "FORBIDDEN");
    }
}

/*
|--------------------------------------------------------------------------
| REQUIRE STAFF (Optional / Future)
|--------------------------------------------------------------------------
*/
function require_staff(): void
{
    require_auth();

    if ($GLOBALS['auth_user']['role'] !== 'staff') {
        sendError("Staff access only", 403, "FORBIDDEN");
    }
}
